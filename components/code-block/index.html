// 从 --moh-* 变量读取主题色（浅色 / 深色都正确）
const css = getComputedStyle(document.documentElement);
const primary = css.getPropertyValue("--moh-primary").trim();

function buildOption(data) {
  return {
    grid: { left: 40, right: 16, top: 24, bottom: 28 },
    xAxis: {
      type: "category",
      data: data.map((d) => d.label),
      axisLabel: { color: css.getPropertyValue("--moh-muted").trim() }
    },
    series: [
      {
        type: "line",
        data: data.map((d) => d.value),
        lineStyle: { color: primary, width: 2 },
        itemStyle: { color: primary }
      }
    ]
  };
}

提示：高亮 token 配色全部用 --moh-* 语义变量（主色=关键词、成功色=字符串、警告色=数字…），任意主题下都协调；复制按钮给 aria-label 并反馈"已复制"。
